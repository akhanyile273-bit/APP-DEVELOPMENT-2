const navToggle = document.getElementById('navToggle');
const primaryNav = document.getElementById('primaryNav');

if (navToggle && primaryNav) {
    navToggle.addEventListener('click', () => {
        const isOpen = primaryNav.classList.toggle('nav-open');
        navToggle.classList.toggle('active', isOpen);

        // aria-expanded tells screen readers whether the menu is open —
        // good practice for accessibility, costs us one line of code.
        navToggle.setAttribute('aria-expanded', isOpen);
    });
}

const header = document.querySelector('.safe-lens-header');

function updateHeaderShadow() {
    if (!header) return;
    header.classList.toggle('scrolled', window.scrollY > 10);
}

window.addEventListener('scroll', updateHeaderShadow);
updateHeaderShadow(); // run once immediately, in case the page loads mid-scroll


const revealTargets = document.querySelectorAll('.intro p');

revealTargets.forEach((el) => el.classList.add('reveal'));

const revealObserver = new IntersectionObserver((entries) => {
    entries.forEach((entry) => {
        if (entry.isIntersecting) {
            entry.target.classList.add('visible');
            revealObserver.unobserve(entry.target); // only animate once
        }
    });
}, {
    threshold: 0.2, // trigger once 20% of the element is visible
});

revealTargets.forEach((el) => revealObserver.observe(el));


const floatersContainer = document.getElementById('floaters');
const emojiChoices = ['👍', '❤️', '😂', '🛡️', '📸', '✅'];
const numberOfFloaters = 10;

if (floatersContainer) {
    for (let i = 0; i < numberOfFloaters; i++) {
        const span = document.createElement('span');
        span.className = 'floater';
        span.textContent = emojiChoices[Math.floor(Math.random() * emojiChoices.length)];

        // Random horizontal position across the hero (0% - 95%)
        span.style.left = `${Math.random() * 95}%`;
        // Random vertical starting point so they don't all line up
        span.style.top = `${Math.random() * 90}%`;
        // Random animation speed (6s - 12s) and start delay, so the
        // floating never looks perfectly synced/robotic
        span.style.animationDuration = `${6 + Math.random() * 6}s`;
        span.style.animationDelay = `${Math.random() * 4}s`;

        floatersContainer.appendChild(span);
    }
}
